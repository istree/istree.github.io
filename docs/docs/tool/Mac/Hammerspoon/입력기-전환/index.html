한영 전환을 Hammerspoon 으로 직접 한다. 오른쪽 Cmd 로 영어와 한국어를 오가고, Ctrl+Space 로 영어와 짝지을 언어를 한국어 · 일본어 중에 바꾸고, Esc 를 누르면 영어로 돌아간다. macOS 기본 단축키로 바꾸면 바뀌기까지 틈이 있어 첫 글자가 이전 언어로 들어가는 일이 있는데, hs.keycodes.currentSourceID 로 바꾸면 바로 바뀐다.

준비

  • 「시스템 설정 › 키보드 › 텍스트 입력 › 입력 소스」 에 ABC · 2벌식 · 일본어(로마자 입력) 를 넣는다
  • 같은 곳에서 Caps Lock 으로 ABC 와 오가는 설정을 끄고, 「키보드 단축키 › 입력 소스」 의 「이전 입력 소스 선택」(Ctrl+Space) 도 끈다. 켜 두면 Hammerspoon 의 단축키와 겹친다
  • 오른쪽 Cmd 는 hidutil 로 F18 로 바꿔 둔다 (remapper:remap('rcmd', 'f18'))

입력기 ID

hs.keycodes.currentSourceID()     -- "com.apple.inputmethod.Korean.2SetKorean"
hs.keycodes.layouts(true)         -- { "com.apple.keylayout.ABC" }
hs.keycodes.methods(true)         -- { "com.apple.inputmethod.Korean.2SetKorean",
                                  --   "com.apple.inputmethod.Kotoeri.RomajiTyping.Japanese", … }
hs.keycodes.currentSourceID("com.apple.keylayout.ABC")   -- 바꾼다. 되면 true
  • 영어처럼 글자를 그대로 내는 것은 자판(layout), 한국어 · 일본어처럼 글자를 조합하는 것은 입력기(method) 로 따로 나온다
  • true 를 빼면 「ABC」 · 「2-Set Korean」 같은 이름이 나온다. 바꿀 때는 ID 를 쓴다

한영 · 한일

local inputEnglish = "com.apple.keylayout.ABC"
local inputKorean = "com.apple.inputmethod.Korean.2SetKorean"
local inputJapanese = "com.apple.inputmethod.Kotoeri.RomajiTyping.Japanese"

local lastSource = inputKorean

-- F18(오른쪽 Cmd) — 영어 ↔ 마지막 언어
local function toggleABC()
	local current = hs.keycodes.currentSourceID()
	if (current == inputEnglish) then
		hs.keycodes.currentSourceID(lastSource)
	else
		lastSource = current
		hs.keycodes.currentSourceID(inputEnglish)
	end
end

-- Ctrl+Space — 짝지을 언어를 한국어 ↔ 일본어로 바꾸고 그 언어로
local function toggleInputSource()
	if (lastSource == inputKorean) then
		lastSource = inputJapanese
	else
		lastSource = inputKorean
	end
	hs.keycodes.currentSourceID(lastSource)
end

hs.hotkey.bind({}, 'f18', toggleABC)
hs.hotkey.bind({'control'}, 'space', toggleInputSource)
  • 영어로 바꿀 때 지금 언어를 lastSource 에 넣는다. 메뉴 막대에서 일본어로 바꾼 뒤에도 F18 두 번이면 일본어로 돌아온다
  • hs.hotkey 는 모든 앱에서 그 키를 가로챈다. 터미널 · emacs 에서 Ctrl+Space 를 쓰는 기능은 동작하지 않는다

Esc 로 영어

vim 처럼 Esc 로 모드를 바꾸는 곳에서 한글이 그대로 남아 명령이 먹지 않는 것을 막는다.

local esc_bind

local function convert_to_eng_with_esc()
	-- 단축키를 끄고 Esc 를 보낸 뒤 다시 켠다
	esc_bind:disable()
	hs.eventtap.keyStroke({}, 'escape', 20000)
	esc_bind:enable()

	if (hs.keycodes.currentSourceID() ~= inputEnglish) then
		hs.keycodes.currentSourceID(inputEnglish)
	end
end

esc_bind = hs.hotkey.new({}, 'escape', convert_to_eng_with_esc):enable()
  • Esc 에 단축키를 걸면 Esc 가 앱에 가지 않는다. 단축키를 잠깐 끄고 keyStroke 로 Esc 를 직접 보낸다
  • keyStroke 의 세 번째 인자는 키를 누르고 뗄 때까지의 지연(µs) 이다. 기본값 200ms 로 두면 Esc 가 느리게 먹는다. 0 으로 하면 보낸 Esc 가 처리되기 전에 단축키가 다시 켜져 자기가 보낸 Esc 를 또 잡을 수 있어, 20ms 로 둔다
  • Esc 를 보내면 조합 중인 한글이 확정된 뒤 영어로 바뀐다

전체 init.lua

require('hs.ipc')

-- .lua 를 저장하면 다시 불러온다
local function reloadOnLua(paths)
	for _, path in ipairs(paths) do
		if path:sub(-4) == '.lua' then
			hs.reload()
			return
		end
	end
end

local realConfigDir = hs.fs.pathToAbsolute(hs.configdir .. '/init.lua'):match('(.*)/')
configWatchers = { hs.pathwatcher.new(hs.configdir, reloadOnLua):start() }
if realConfigDir ~= hs.configdir then
	table.insert(configWatchers, hs.pathwatcher.new(realConfigDir, reloadOnLua):start())
end

-- 키 바꾸기 — Caps Lock ↔ 왼쪽 Ctrl, 오른쪽 Cmd → F18
local FRemap = require('foundation_remapping')
local remapper = FRemap.new()

remapper:remap('lctrl', 'capslock')
remapper:remap('capslock', 'lctrl')
remapper:remap('rcmd', 'f18')
remapper:register()
