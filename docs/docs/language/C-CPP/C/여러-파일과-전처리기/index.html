프로그램이 커지면 기능마다 .c 파일을 나누고, 밖에 보일 선언을 .h 헤더에 둔다. .c 는 하나씩 따로 컴파일되고(번역 단위), 링커가 마지막에 이어 붙인다. #include · #define 은 컴파일 전에 전처리기가 글자 그대로 바꿔 넣는 것이다.

헤더와 소스

include/counter.h
src/counter.c
src/main.c
// include/counter.h — 밖에 보일 것만
#ifndef COUNTER_H              // include guard — 두 번 들어와도 한 번만 읽는다
#define COUNTER_H

#include <stddef.h>

#define COUNTER_MAX 100

extern int counter_total;      // 선언 — 어딘가에 있다 (정의는 counter.c)

void counter_add(int n);
int counter_get(void);
size_t counter_calls(void);

static inline int clamp(int v, int lo, int hi) {   // 작은 함수는 헤더에 inline 으로
    return v < lo ? lo : v > hi ? hi : v;
}

#endif
// src/counter.c
#include "counter.h"

int counter_total = 0;         // 정의 — 한 곳에만
static size_t calls = 0;       // static — 이 파일 안에서만 보인다

static int limit(int v) { return clamp(v, 0, COUNTER_MAX); }

void counter_add(int n) { calls++; counter_total = limit(counter_total + n); }
int counter_get(void) { return counter_total; }
size_t counter_calls(void) { return calls; }
// src/main.c
#include <stdio.h>
#include "counter.h"           // "" — 내 헤더, <> — 시스템 헤더

int main(void) {
    counter_add(60);
    counter_add(70);
    printf("%d %zu\n", counter_get(), counter_calls());   // 100 2
}
clang -std=c17 -Wall -Wextra -Iinclude -o app src/main.c src/counter.c   # 한 번에
./app

clang -std=c17 -Iinclude -c src/counter.c -o counter.o   # 따로 컴파일 — 바뀐 파일만 다시
clang -std=c17 -Iinclude -c src/main.c -o main.o
clang main.o counter.o -o app                            # 링크

ar rcs libcounter.a counter.o          # 정적 라이브러리로 묶기
clang main.o -L. -lcounter -o app      # libcounter.a 를 링크
  • #pragma once 를 맨 위에 두면 include guard 와 같다. 표준은 아니지만 clang · gcc · MSVC 가 모두 지원한다
  • 파일이 여럿이면 직접 명령을 늘어놓지 않고 CMake · make 로 빌드한다

static · extern

쓰는 곳 static extern
파일 맨 위의 변수 · 함수 그 파일 안에서만 보인다 다른 파일에 있는 것을 선언한다
함수 안의 변수 함수가 끝나도 값이 남는다 (한 번만 초기화) —
  • 다른 파일에서 static 인 calls 를 쓰면 error: use of undeclared identifier 'calls' 로 컴파일되지 않는다. 숨길 것은 static 으로 두고, 필요하면 함수로 내준다
  • 헤더에 변수를 정의(int total = 0;) 하면 그 헤더를 include 한 파일마다 하나씩 생겨 링크 오류가 난다. 헤더에는 extern 선언만 둔다

링크 오류

선언만 있고 정의가 없으면 컴파일은 되고 링크에서 멈춘다.

Undefined symbols for architecture arm64:
  "_counter_add", referenced from:
      _main in y-91dc36.o
ld: symbol(s) not found for architecture arm64
clang: error: linker command failed with exit code 1 (use -v to see invocation)
  • 그 함수가 들어 있는 .c 를 같이 넘기지 않았거나, 라이브러리(-lm · -lcounter) 를 빠뜨렸거나, 이름 · 인자가 선언과 정의에서 다를 때 난다
  • gcc · 리눅스에서는 undefined reference to 'counter_add' 로 나온다

매크로

#define COUNTER_MAX 100                 // 상수
#define SQUARE(x) ((x) * (x))           // 함수처럼 — 인자와 전체를 괄호로
#define BAD_SQUARE(x) x * x

SQUARE(1 + 2);        // ((1 + 2) * (1 + 2)) = 9
BAD_SQUARE(1 + 2);    // 1 + 2 * 1 + 2 = 5 — 글자 그대로 바뀐다

#ifdef DEBUG                            // clang -DDEBUG 로 켠다
#define LOG(...) fprintf(stderr, "[debug] " __VA_ARGS__)
#else
#define LOG(...) ((void)0)
#endif
LOG("total=%d\n", counter_get());

#if COUNTER_MAX > 50
/* 조건부 컴파일 */
#endif

printf("%s:%d %s\n", __FILE__, __LINE__, __func__);   // src/main.c:16 main
  • 전처리 결과는 clang -E src/main.c 로 본다. 매크로가 이상하게 펼쳐질 때 확인한다
  • 인자를 두 번 쓰는 매크로에 SQUARE(i++) 를 넘기면 i 가 두 번 늘어난다. 계산은 static inline 함수로, 상수는 enum · (C23) constexpr 로 쓰는 편이 안전하다

경고 · 검사 옵션

clang -std=c17 -Wall -Wextra -Wpedantic -Wshadow -Wconversion -g -fsanitize=address,undefined …
옵션 하는 일
-Wall -Wextra 흔한 실수 경고 (늘 켠다)
-Wpedantic 표준 밖의 확장 문법 경고
-Wshadow · -Wconversion 바깥 이름 가리기 · 값이 잘리는 타입 바꾸기
-Werror 경고를 오류로 — CI 에서
-g · -O2 디버그 정보 · 최적화
-fsanitize=address,undefined 메모리 · 정의되지 않은 동작을 실행 중에 잡는다
