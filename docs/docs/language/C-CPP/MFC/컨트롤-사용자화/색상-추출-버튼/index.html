버튼을 누른 채 끌면 화면 어디든 커서 아래의 색을 집어 온다. CButton 을 상속해 만든다.

마우스 붙잡기

누르는 순간 SetCapture 로 마우스를 붙잡으면 커서가 창 밖으로 나가도 WM_MOUSEMOVE 를 계속 받는다.

// ColorPickButton.cpp — CButton 파생
BEGIN_MESSAGE_MAP(CColorPickButton, CButton)
    ON_WM_LBUTTONDOWN()
    ON_WM_MOUSEMOVE()
    ON_WM_LBUTTONUP()
END_MESSAGE_MAP()

void CColorPickButton::OnLButtonDown(UINT nFlags, CPoint point)
{
    SetCapture();
    m_bTracking = TRUE;
    ::SetCursor(m_hCursor);   // 스포이트 커서
    CButton::OnLButtonDown(nFlags, point);
}

void CColorPickButton::OnMouseMove(UINT nFlags, CPoint point)
{
    if (m_bTracking)
        PickColor();
    CButton::OnMouseMove(nFlags, point);
}

void CColorPickButton::OnLButtonUp(UINT nFlags, CPoint point)
{
    if (m_bTracking) {
        ReleaseCapture();
        m_bTracking = FALSE;
    }
    CButton::OnLButtonUp(nFlags, point);
}

마우스를 붙잡고 있는 동안에는 WM_SETCURSOR 가 오지 않아 커서가 원래대로 돌아가지 않는다. 그래서 누를 때 한 번만 바꾸면 된다.

화면의 색 읽기

화면 전체의 DC 에서 커서 자리의 점을 읽는다.

void CColorPickButton::PickColor()
{
    POINT pt;
    ::GetCursorPos(&pt);

    HDC hdc = ::GetDC(NULL);                      // 화면 전체
    COLORREF color = ::GetPixel(hdc, pt.x, pt.y);
    ::ReleaseDC(NULL, hdc);                       // 끌 때마다 불리므로 꼭 돌려준다

    if (m_pReceiver)
        m_pReceiver->PostMessage(m_nMessage, (WPARAM)color);
}

받을 창 정하기

받을 창과 메시지를 바깥에서 정해 주면 어느 대화상자에서나 쓸 수 있다.

// ColorPickButton.h
void SetReceiver(CWnd* pReceiver, UINT nMessage) { m_pReceiver = pReceiver; m_nMessage = nMessage; }
void SetCursor(HCURSOR hCursor)                  { m_hCursor = hCursor; }
// 쓰는 대화상자
#define WM_COLORPICKED (WM_APP + 10)

BOOL CMyDlg::OnInitDialog()
{
    CDialog::OnInitDialog();
    m_btnPick.SetReceiver(this, WM_COLORPICKED);
    m_btnPick.SetCursor(AfxGetApp()->LoadCursor(IDC_CURSOR_PICK));
    return TRUE;
}

ON_MESSAGE(WM_COLORPICKED, &CMyDlg::OnColorPicked)

LRESULT CMyDlg::OnColorPicked(WPARAM wParam, LPARAM)
{
    m_pModel->SetColor((COLORREF)wParam);
    UpdatePreview();   // 끄는 동안 계속 오므로 미리보기가 따라 바뀐다
    return 0;
}

스포이트 커서를 리소스에 넣는 법은 리소스 에 정리했다.
