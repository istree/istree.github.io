CWnd 를 바로 상속해 처음부터 그리는 컨트롤이다. 표준 컨트롤에 없는 것을 보여 주고 고르게 할 때 쓴다. 범례를 차트의 어느 쪽에 둘지 그림으로 보여 주는 창 같은 것이다.

만들기

CWnd 를 상속하고, 템플릿에 자리만 잡아 둔 Picture Control 위치에 Create 로 만든다.

// PositionWnd.h
class CPositionWnd : public CWnd
{
public:
    void SetPosition(int nPos);
    int  GetPosition() const { return m_nPos; }
protected:
    afx_msg void OnPaint();
    afx_msg BOOL OnEraseBkgnd(CDC* pDC);
    afx_msg void OnLButtonDown(UINT nFlags, CPoint point);
    DECLARE_MESSAGE_MAP()
    int     m_nPos = 0;
    CBitmap m_bmp[4];   // 왼쪽 · 위 · 오른쪽 · 아래
};
// 쓰는 대화상자의 OnInitDialog
CRect rc;
GetDlgItem(IDC_POSITION_AREA)->GetWindowRect(&rc);
ScreenToClient(&rc);
m_wndPosition.Create(NULL, NULL, WS_CHILD | WS_VISIBLE, rc, this, IDC_POSITION);

값과 그리기

값이 바뀌면 Invalidate 로 다시 그리게 하고, 그리는 일은 OnPaint 에서 한다. 위치마다 비트맵을 하나씩 두고 지금 값에 맞는 것을 찍는다.

void CPositionWnd::SetPosition(int nPos)
{
    m_nPos = nPos;
    Invalidate();
}

void CPositionWnd::OnPaint()
{
    CPaintDC dc(this);
    CDC mem;
    mem.CreateCompatibleDC(&dc);
    CBitmap* pOld = mem.SelectObject(&m_bmp[m_nPos]);

    CRect rc;
    GetClientRect(&rc);
    dc.BitBlt(0, 0, rc.Width(), rc.Height(), &mem, 0, 0, SRCCOPY);

    mem.SelectObject(pOld);
}

BOOL CPositionWnd::OnEraseBkgnd(CDC* pDC)
{
    return TRUE;   // 그림이 창을 다 덮으므로 바탕을 지우지 않는다 (깜빡임 없음)
}

비트맵은 Create 를 오버라이드하거나 처음 쓰기 전에 LoadBitmap 으로 불러 둔다. 깜빡임 없이 그리는 법은 더블 버퍼링 에 정리했다.

마우스로 고르기

누른 자리로 값을 정하고 부모에게 알린다.

void CPositionWnd::OnLButtonDown(UINT nFlags, CPoint point)
{
    CRect rc;
    GetClientRect(&rc);
    int nPos = (point.x < rc.Width() / 3) ? 0 : (point.x > rc.Width() * 2 / 3) ? 2
             : (point.y < rc.Height() / 2) ? 1 : 3;
    SetPosition(nPos);
    GetParent()->SendMessage(WM_POSITIONCHANGED, GetDlgCtrlID(), nPos);
    CWnd::OnLButtonDown(nFlags, point);
}

부모가 받는 사용자 메시지는 메시지 에 정리했다.
