왼쪽 트리에서 대상을 고르면, 오른쪽 탭에 그 대상의 페이지들이 나오는 속성 창이다. 대상이 많고 대상마다 고칠 것이 여러 묶음일 때 쓴다.

설계

화면(트리 · 탭 · 페이지) 과 "어느 대상을 어느 페이지로 보일지" 를 나눈다. 그 사이를 매핑 객체가 연결한다.

  • 매핑 객체(TabPageInfo) — 탭 이름 + 페이지 다이얼로그 ID + 고칠 대상(프로퍼티)
  • 트리 항목 정보(TreeItemInfo) — 트리 항목 하나에 달린 매핑 객체 목록
  • 커맨더(CTreeTabCommander) — 트리 항목을 만들고, 경로로 찾아 고른다
  • 페이지 — 다이얼로그 ID 마다 하나만 만들어 두고, 매핑 객체를 넘겨받아 값을 읽고 쓴다

매핑 객체

매핑 객체는 탭 이름 · 페이지 다이얼로그 ID · 고칠 대상을 묶는다. 대상이 하나인 페이지는 템플릿 하나로 만들고, 페이지마다 typedef 로 이름을 붙인다.

// TreeTabMapping.h
class TabPageInfo
{
public:
    virtual ~TabPageInfo() {}            // 목록에서 지울 때 파생 객체까지
    UINT    GetID()   const { return m_nPageID; }
    CString GetName() const { return m_strName; }
    TabPagePath m_path;                  // 트리 번호 · 트리 항목 · 탭 위치 (AddTabPage 가 채운다)
protected:
    UINT    m_nPageID;
    CString m_strName;
};

template<typename PROP, UINT PAGE_ID>
class TabPageSingle : public TabPageInfo
{
public:
    TabPageSingle(LPCTSTR name, PROP* pProp) : m_pProp(pProp) { m_nPageID = PAGE_ID; m_strName = name; }
    PROP* GetProp() { return m_pProp; }
private:
    PROP* m_pProp;
};

typedef TabPageSingle<IFontProp,   IDD_PAGE_FONT>   TabPageFontInfo;     // 글꼴
typedef TabPageSingle<ILegendProp, IDD_PAGE_LEGEND> TabPageLegendInfo;   // 범례
typedef TabPageSingle<IAxisProp,   IDD_PAGE_AXIS>   TabPageAxisInfo;     // 축

대상이 둘 이상인 페이지는 따로 클래스를 둔다. 배경과 테두리를 한 페이지에서 고치는 식이다.

class TabPageFillLineInfo : public TabPageInfo
{
public:
    TabPageFillLineInfo(LPCTSTR name, IFillProp* pFill, ILineProp* pLine)
        : m_pFill(pFill), m_pLine(pLine) { m_nPageID = IDD_PAGE_FILLLINE; m_strName = name; }
    IFillProp* m_pFill;
    ILineProp* m_pLine;
};
  • 같은 글꼴 페이지를 제목 · 범례 · 축 제목이 같이 쓴다. 매핑 객체만 다르다
  • 대상은 모두 인터페이스다. 데이터 인터페이스 에 정리했다

트리 항목 정보

트리 항목 하나에 매핑 객체 목록을 단다. 붙일 때 매핑 객체에 자기 경로(어느 트리 · 어느 항목 · 몇 번째 탭) 를 적어 준다.

class TreeItemInfo
{
public:
    void AddTabPage(TabPageInfo* pPage)
    {
        pPage->m_path = m_path;
        pPage->m_path.nTab = (int)m_pages.size();   // 몇 번째 탭인지
        m_pages.push_back(pPage);
    }
    size_t       GetCount() const  { return m_pages.size(); }
    TabPageInfo* At(int n)         { return m_pages[n]; }
    TabPagePath  m_path;
    ~TreeItemInfo() { for (auto p : m_pages) delete p; }
private:
    std::vector<TabPageInfo*> m_pages;
};

계층으로 쌓기

초기화할 때 트리 항목을 위에서부터 만들고, 항목마다 매핑 객체를 붙인다. 일종의 빌더 패턴이다. 화면 코드는 이 구성만 읽고 그린다.

void CSeriesPanel::BuildTree()
{
    // 부모: 시리즈 전체
    TreeItemInfo* pAll = m_pCommander->CreateItemParent(_T("시리즈 전체"));
    pAll->AddTabPage(new TabPageFrameInfo(_T("시리즈 모양"), pChart->GetFrame()));
    pAll->AddTabPage(new TabPagePaletteInfo(_T("팔레트"), pChart->GetPalette()));

    // 자식: 계열마다, 계열 종류에 따라 다른 페이지
    for (int i = 0; i < pChart->GetSeriesCount(); i++) {
        ISeriesProp* pSeries = pChart->GetSeries(i);
        TreeItemInfo* pItem = m_pCommander->CreateItemChild(pAll, pSeries->GetName());

        pItem->AddTabPage(new TabPageSeriesCommonInfo(_T("공통"), pSeries));      // 모든 계열
        pItem->AddTabPage(new TabPageMarkInfo(_T("마크"), pSeries->GetMark()));

        switch (pSeries->GetType()) {
        case SERIES_BAR:
            ADD_TABPAGE_FILLLINE(_T("배경 · 테두리"), pItem, pSeries);
            pItem->AddTabPage(new TabPageBarInfo(_T("막대"), pSeries));
            break;
        case SERIES_LINE:
            pItem->AddTabPage(new TabPageLineInfo(_T("선"), pSeries));
            break;
        // ...
        }
    }
}

여러 계열이 같은 모양으로 붙이는 페이지는 매크로로 줄인다.

#define ADD_TABPAGE_FILLLINE(NAME, ITEM, PROP) \
    (ITEM)->AddTabPage(new TabPageFillLineInfo(NAME, (PROP)->GetFill(), (PROP)->GetLine()))

탭에 보이기

트리에서 항목을 고르면 그 항목의 매핑 목록으로 탭을 다시 채운다. 페이지는 다이얼로그 ID 로 찾아, 없으면 그때 만들고, 매핑 객체를 넘긴다.

void CPropTab::Init(TreeItemInfo* pItem)
{
    HideAllPages();
    DeleteAllItems();
    for (int i = 0; i < (int)pItem->GetCount(); i++) {
        TabPageInfo* pInfo = pItem->At(i);
        CPropPage* pPage = m_pages[pInfo->GetID()];      // 다이얼로그 ID 마다 하나
        if (!::IsWindow(pPage->GetSafeHwnd()))
            pPage->Create(pInfo->GetID(), this);         // 처음 쓸 때 만든다
        InsertItem(i, pInfo->GetName());
    }
    ShowPage(0);
}

void CPropTab::ShowPage(int nTab)
{
    TabPageInfo* pInfo = m_pItem->At(nTab);
    CPropPage* pPage = m_pages[pInfo->GetID()];
    pPage->SetTabPageInfo(pInfo);    // 대상을 넘긴다
    pPage->LoadProp();               // 대상의 값을 컨트롤에

    CRect rc;
    GetClientRect(&rc);
    AdjustRect(FALSE, &rc);          // 탭 머리를 뺀 안쪽
    pPage->SetWindowPos(NULL, rc.left, rc.top, rc.Width(), rc.Height(), SWP_NOZORDER | SWP_SHOWWINDOW);
}

페이지는 탭의 자식 창(패널 윈도우) 이다. 만드는 법은 대화상자 의 "패널 윈도우" 에 정리했다.

페이지에서 대상 받기

페이지는 자기 다이얼로그 ID 에 맞는 매핑 객체가 온다는 것을 안다. 받은 매핑 객체에서 대상을 꺼내 읽고 쓴다.

void CFontPage::LoadProp()
{
    IFontProp* pFont = static_cast<TabPageFontInfo*>(m_pInfo)->GetProp();
    m_strName = pFont->GetName();
    m_nSize   = pFont->GetSize();
    UpdateData(FALSE);
}

경로로 찾아가기

매핑 객체마다 경로(트리 번호 · 트리 항목 · 탭 위치) 가 있으므로, 다른 곳에서 특정 페이지로 바로 보낼 수 있다. 미리보기에서 범례를 누르면 범례 › 글꼴 페이지를 여는 식이다.

void CTreeTabCommander::Select(const TabPagePath& path)
{
    CTreeCtrl* pTree = m_trees[path.nTree];
    pTree->SelectItem(path.hItem);        // TVN_SELCHANGED 로 탭이 다시 채워진다
    GetTab(path.nTree)->ShowPage(path.nTab);
}
