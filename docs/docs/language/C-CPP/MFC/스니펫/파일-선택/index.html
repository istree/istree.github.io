파일 열기 · 저장 대화상자로 경로를 받는다.

CFileDialog dlg(TRUE, _T("xml"), NULL,                   // TRUE = 열기
                OFN_HIDEREADONLY | OFN_FILEMUSTEXIST,
                _T("XML 파일 (*.xml)|*.xml|모든 파일 (*.*)|*.*||"), this);
if (dlg.DoModal() == IDOK) {
    CString path = dlg.GetPathName();
}
  • 필터는 설명|패턴| 을 이어 쓰고 끝에 ||
  • 저장은 첫 인자를 FALSE 로, 덮어쓸 때 묻도록 OFN_OVERWRITEPROMPT 를 준다. 두 번째 인자(xml) 는 확장자를 안 쓰면 붙여 줄 기본 확장자다

여러 파일 고르기:

CFileDialog dlg(TRUE, NULL, NULL, OFN_ALLOWMULTISELECT | OFN_FILEMUSTEXIST, _T("모든 파일 (*.*)|*.*||"), this);
CString buf;
dlg.GetOFN().lpstrFile = buf.GetBuffer(64 * 1024);   // 이름이 많으면 기본 버퍼가 모자란다
dlg.GetOFN().nMaxFile  = 64 * 1024;
if (dlg.DoModal() == IDOK) {
    POSITION pos = dlg.GetStartPosition();
    while (pos)
        AddFile(dlg.GetNextPathName(pos));
}
buf.ReleaseBuffer();
