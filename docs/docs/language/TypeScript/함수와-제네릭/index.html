함수는 매개변수에 타입을 적고, 반환 타입은 보통 추론에 맡긴다. 여러 타입에 같은 코드를 쓰면서 타입 정보를 잃지 않으려면 제네릭(타입 매개변수) 을 쓴다.

함수 타입

function add(a: number, b: number): number {      // 반환 타입은 적어도 되고 빼도 된다
  return a + b;
}
const mul = (a: number, b: number) => a * b;      // number 로 추론

function greet(name: string, greeting = "Hello", suffix?: string) {   // 기본값 · 선택 매개변수
  return `${greeting}, ${name}${suffix ?? ""}`;
}
greet();             // TS2554: Expected 1-3 arguments, but got 0.

function sum(...nums: number[]) {                 // 나머지 매개변수
  return nums.reduce((s, n) => s + n, 0);
}

함수를 값으로 넘길 때는 함수 타입에 이름을 붙인다.

type Handler = (event: { type: string }) => void;
const onClick: Handler = e => console.log(e.type);   // e 의 타입을 적지 않아도 된다

type Callback = (err: Error | null, data?: string) => void;
  • 밖으로 내보내는 함수(라이브러리 · 모듈 경계) 는 반환 타입을 적는다. 안쪽을 고쳤을 때 바깥 타입이 몰래 바뀌는 것을 막는다
  • 반환 값이 없으면 void, 늘 던지면 never 다

오버로드

인자 타입에 따라 반환 타입이 다르면 선언을 여러 개 적고, 마지막에 구현을 하나 둔다. 부르는 쪽은 선언만 본다.

function parse(input: string): number;
function parse(input: string[]): number[];
function parse(input: string | string[]) {
  return Array.isArray(input) ? input.map(Number) : Number(input);
}

const one = parse("1");           // number
const many = parse(["1", "2"]);   // number[]
  • 갈래가 둘뿐이면 유니온 매개변수 하나로 충분한지 먼저 본다. 오버로드는 구현 안의 타입 검사가 느슨해진다

제네릭

<T> 는 부를 때 정해지는 타입이다. 대개 인자에서 추론하므로 부르는 쪽은 적지 않는다.

function first<T>(list: T[]): T | undefined {
  return list[0];
}
first([1, 2]);       // number | undefined
first(["a"]);        // string | undefined

async function fetchJson<T>(url: string): Promise<T> {
  const res = await fetch(url);
  return res.json() as Promise<T>;       // 서버가 정말 그 모양인지는 검사하지 않는다
}
type Todo = { id: number; title: string };
const todos = await fetchJson<Todo[]>("/api/todos");   // 추론할 인자가 없으면 적어 준다

extends 로 T 에 올 수 있는 타입을 제한한다.

function longest<T extends { length: number }>(a: T, b: T): T {
  return a.length >= b.length ? a : b;
}
longest("ab", "c");      // 된다 — string 에 length 가 있다
longest([1], [2, 3]);    // 된다
longest(1, 2);           // TS2345: Argument of type 'number' is not assignable to parameter of type '{ length: number; }'.

function pick<T, K extends keyof T>(obj: T, key: K): T[K] {   // key 는 obj 의 키 중 하나
  return obj[key];
}
const user = { id: 1, name: "geon" };
pick(user, "name");      // string
pick(user, "age");       // TS2345: Argument of type '"age"' is not assignable to parameter of type '"id" | "name"'.
  • 제네릭은 입력과 출력의 타입을 이을 때 쓴다. T 가 한 곳에만 나오면 그냥 그 타입(unknown 등) 을 쓴다

제네릭 타입 · 클래스도 같다.

type ApiResponse<T = unknown> = { ok: boolean; data: T };   // 기본값
const res: ApiResponse<Todo[]> = { ok: true, data: [] };

class Box<T> {
  value: T;
  constructor(value: T) { this.value = value; }
  map<U>(fn: (v: T) => U): Box<U> { return new Box(fn(this.value)); }
}
new Box(2).map(n => `${n}`);   // Box<string>

조건부 타입

T extends X ? A : B 는 타입에서 하는 삼항 연산이다. infer 로 안쪽 타입을 꺼낸다. 라이브러리 타입에 많고, 앱 코드에서는 읽을 줄 알면 된다.

type IsString<T> = T extends string ? "yes" : "no";
type A = IsString<"a">;          // "yes"
type B = IsString<1>;            // "no"

type ElementOf<T> = T extends (infer E)[] ? E : never;
type E = ElementOf<string[]>;    // string

type Unwrap<T> = T extends Promise<infer V> ? V : T;
type V = Unwrap<Promise<number>>;   // number — 표준의 Awaited<T> 가 이것이다
