테스트는 같은 파일 안 tests 모듈에 #[test] 함수로 쓰고, cargo test 로 돌린다.

테스트 쓰기

pub fn add_two(a: i32) -> i32 {
    a + 2
}

#[cfg(test)]                       // cargo test 할 때만 컴파일
mod tests {
    use super::*;

    #[test]
    fn it_adds_two() {
        assert_eq!(add_two(2), 4);
    }

    #[test]
    fn greeting_has_name() {
        let result = format!("Hello {}", "Carol");
        assert!(result.contains("Carol"), "이름이 없다: {result}");   // 실패 메시지
    }

    #[test]
    #[should_panic(expected = "0 으로")]   // panic 해야 통과
    fn divide_by_zero() {
        panic!("0 으로 나눴다");
    }

    #[test]
    fn with_result() -> Result<(), String> {   // Err 를 돌려주면 실패
        if add_two(2) == 4 { Ok(()) } else { Err(String::from("틀렸다")) }
    }
}

돌리기

cargo test                          # 모두
cargo test add                      # 이름에 add 가 든 것만
cargo test -- --test-threads=1      # 차례로 (파일 · 전역 상태를 같이 쓸 때)
cargo test -- --show-output         # 통과한 테스트의 println! 도 보기
cargo test -- --ignored             # #[ignore] 를 붙여 둔 느린 테스트만
  • -- 뒤는 테스트 실행 파일에 넘기는 옵션이다
  • 통합 테스트는 tests/ 폴더에 파일로 둔다. 공개된 API 만 쓸 수 있다
