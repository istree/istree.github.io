jQuery.when 에서 배열 사용

여러개의 url 응답이 모두 완료된 후 뭔가를 실행해야 할 경우 when과 함께 Deferred Object를 사용한다.

예를 들어 두개의 url의 응답을 받아 결과를 출력하려면 다음과 같다.

jQuery.when( 
    jQuery.get('a.txt'), 
    jQuery.get('b.txt') 
).done( function(a,b) {
    console.log(a);
    console.log(b);
});

다만 when의 파라미터는 고정되어 있고 배열을 처리해주지 않는다.

요청할 url이 여러개일 경우 배열로 처리하고 싶다면? 아래와 같이 할 수 있다.

var acts = urls.map(function(url) {
    return jQuery.get(url);
});

jQuery.when.apply(
    jQuery,
    acts
).done(function() {
    var results = Array.prototype.slice.call(arguments);
    results.map(function(result) {
        console.log(result[0]);
    });
});

관련글 https://stackoverflow.com/questions/5627284/pass-in-an-array-of-deferreds-to-when

지금 문법으로는 apply 대신 전개(...) 를 쓴다.

jQuery.when(...urls.map(url => jQuery.get(url)))
  .done((...results) => {
    for (const [data] of results) console.log(data);   // 결과마다 [data, status, xhr]
  });
  • url 이 하나뿐이면 done 이 [data, status, xhr] 배열이 아니라 data, status, xhr 를 따로 받는다. 개수가 바뀌는 목록이면 아래 Promise.all 이 안전하다

jQuery 3 의 Deferred 는 Promise 와 맞물리므로 Promise.all 로 기다려도 된다. 이때는 data 만 온다.

const results = await Promise.all(urls.map(url => jQuery.get(url)));
results.forEach(data => console.log(data));
  • jQuery 없이 하려면 fetch 를 쓴다: await Promise.all(urls.map(url => fetch(url).then(res => res.text()))). 비동기 에 정리했다
