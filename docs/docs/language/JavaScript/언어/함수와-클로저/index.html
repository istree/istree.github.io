JavaScript 의 함수는 값이다. 변수에 넣고, 인자로 넘기고, 함수에서 돌려준다. 함수 안에서 만든 함수는 바깥 변수를 붙잡고 있다(클로저). 비공개 상태 · 콜백이 모두 여기서 나온다.

함수 만들기

// 선언 — 선언보다 위에서 불러도 된다(호이스팅)
function add(a, b) { return a + b; }

// 표현식
const sub = function (a, b) { return a - b; };

// 화살표 — 한 식이면 return 을 뺀다
const mul = (a, b) => a * b;
const point = (x, y) => ({ x, y });   // 객체는 괄호로 감싼다

매개변수

function draw(x, y = 0, { color = "black", width = 1 } = {}) { ... }  // 기본값 · 옵션 객체
draw(10);
draw(10, 20, { color: "red" });

function sum(...nums) {                 // 나머지 매개변수 — 진짜 배열이다
  return nums.reduce((s, n) => s + n, 0);
}
sum(1, 2, 3);                           // 6
sum(...[4, 5, 6]);                      // 15 — 배열을 펼쳐서 넘긴다
  • 예전 코드는 arguments(배열 비슷한 객체) 와 fn.apply(null, array) 를 썼다. 지금은 ... 로 바꾼다
  • 인자가 셋을 넘으면 옵션 객체 하나로 받는다. 부르는 쪽에서 이름이 보인다

this

this 는 함수를 어떻게 불렀는지 로 정해진다. 부르는 법은 네 가지다.

부르는 법 this
obj.method() — 메서드로 obj
fn() — 함수로 undefined (strict) · 전역 객체 (옛 방식)
new Fn() — 생성자로 새로 만든 객체
fn.call(obj, a) · fn.apply(obj, [a]) 넘긴 obj

메서드 안에서 만든 보통 함수는 this 를 잃는다. 화살표 함수는 자기 this 가 없어 바깥의 this 를 그대로 쓴다.

const timer = {
  seconds: 0,
  start() {
    // 옛 방식: const self = this; setInterval(function () { self.seconds++; }, 1000);
    setInterval(() => { this.seconds++; }, 1000);   // 바깥(start) 의 this
  },
};

메서드를 떼어서 넘기면(콜백 · 이벤트 핸들러) this 가 끊어진다. bind 로 묶거나 화살표로 감싼다.

button.addEventListener("click", timer.start);              // this 가 button 이 된다
button.addEventListener("click", timer.start.bind(timer));  // 묶는다
button.addEventListener("click", () => timer.start());      // 감싼다

클로저

안쪽 함수는 바깥 함수가 끝난 뒤에도 바깥 변수를 붙잡고 있다. 밖에서는 그 변수에 손댈 수 없으므로 비공개 상태가 된다.

function createIdGenerator(prefix) {
  let next = 1;                      // 밖에서 보이지 않는다
  return {
    next: () => `${prefix}-${String(next++).padStart(4, "0")}`,
    reset: () => { next = 1; },
  };
}

const orderId = createIdGenerator("ORD");
orderId.next();   // "ORD-0001"
orderId.next();   // "ORD-0002"

타이머 · 이벤트처럼 나중에 불리는 함수가 같은 변수를 계속 보는 것도 클로저다. 입력이 멈추고 300 ms 뒤에 한 번만 검색한다.

function debounce(fn, ms) {
  let timer;                         // 부를 때마다 같은 timer 를 본다
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), ms);
  };
}

input.addEventListener("input", debounce(e => search(e.target.value), 300));

반복문 안의 클로저

var 는 반복 전체에 변수 하나라서, 나중에 불리는 함수가 모두 마지막 값을 본다. let 은 반복마다 새 변수를 만든다.

for (var i = 0; i < 3; i++) setTimeout(() => console.log(i));   // 3, 3, 3
for (let i = 0; i < 3; i++) setTimeout(() => console.log(i));   // 0, 1, 2
  • let 이 없던 때는 즉시 실행 함수로 반복마다 값을 붙잡았다: (function (i) { setTimeout(() => console.log(i)); })(i);

즉시 실행 함수

만들자마자 부르는 함수다. 모듈이 없던 때 전역을 더럽히지 않으려고 파일 전체를 감쌌다.

(function () {
  const local = 1;   // 밖에서 보이지 않는다
  ...
})();
  • 지금은 ES 모듈 파일이 그 자체로 범위를 가진다

예외

throw 는 아무 값이나 던질 수 있지만 Error 를 던진다. 호출 스택이 같이 남는다.

function parseAge(text) {
  const n = Number(text);
  if (!Number.isInteger(n) || n < 0) throw new RangeError(`나이가 아니다: ${text}`);
  return n;
}

try {
  parseAge("abc");
} catch (e) {
  console.error(`${e.name}: ${e.message}`);   // RangeError: 나이가 아니다: abc
} finally {
  ...
}

throw new Error("저장 실패", { cause: err });   // 원래 오류를 붙여 다시 던진다
