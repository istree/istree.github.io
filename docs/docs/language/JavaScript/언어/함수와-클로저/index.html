JavaScript 의 함수는 값이다. 변수에 넣고, 인자로 넘기고, 함수에서 돌려준다. 함수 안에서 만든 함수는 바깥 변수를 붙잡고 있다(클로저). 비공개 상태 · 콜백 · 부분 적용이 모두 여기서 나온다.

함수 만들기

// 선언 — 선언보다 위에서 불러도 된다(호이스팅)
function add(a, b) { return a + b; }

// 표현식
const sub = function (a, b) { return a - b; };

// 화살표 — 한 식이면 return 을 뺀다
const mul = (a, b) => a * b;
const point = (x, y) => ({ x, y });   // 객체는 괄호로 감싼다

매개변수

function draw(x, y = 0, { color = "black", width = 1 } = {}) { ... }  // 기본값 · 옵션 객체
draw(10);
draw(10, 20, { color: "red" });

function sum(...nums) {                 // 나머지 매개변수 — 진짜 배열이다
  return nums.reduce((s, n) => s + n, 0);
}
sum(4, 8, 15);
sum(...[16, 23, 42]);                   // 배열을 펼쳐서 넘긴다
  • 예전 코드는 arguments(배열 비슷한 객체) 와 fn.apply(null, array) 를 썼다. 지금은 ... 로 바꾼다
  • 인자가 셋을 넘으면 옵션 객체 하나로 받는다. 부르는 쪽에서 이름이 보인다

this

this 는 함수를 어떻게 불렀는지 로 정해진다. 핵심 가이드는 네 가지 호출 패턴으로 나눈다.

부르는 법 this
obj.method() — 메서드 호출 obj
fn() — 함수 호출 undefined (strict) · 전역 객체 (옛 방식)
new Fn() — 생성자 호출 새로 만든 객체
fn.call(obj, a) · fn.apply(obj, [a]) 넘긴 obj

메서드 안에서 만든 보통 함수는 this 를 잃는다. 화살표 함수는 자기 this 가 없어 바깥의 this 를 그대로 쓴다.

const counter = {
  value: 0,
  double() {
    // 옛 방식: const that = this; function helper() { that.value *= 2; }
    const helper = () => { this.value *= 2; };   // 바깥(double) 의 this
    helper();
  },
};

메서드를 떼어서 넘기면(콜백 · 이벤트 핸들러) this 가 끊어진다. bind 로 묶거나 화살표로 감싼다.

button.addEventListener("click", counter.double);              // this 가 button 이 된다
button.addEventListener("click", counter.double.bind(counter)); // 묶는다
button.addEventListener("click", () => counter.double());       // 감싼다

클로저

안쪽 함수는 바깥 함수가 끝난 뒤에도 바깥 변수를 붙잡고 있다. 밖에서는 그 변수에 손댈 수 없으므로 비공개 상태가 된다.

function serialMaker(prefix = "") {
  let seq = 0;                       // 밖에서 보이지 않는다
  return {
    setPrefix(p) { prefix = String(p); },
    setSeq(s) { seq = s; },
    next() { return prefix + seq++; },
  };
}

const ids = serialMaker("Q");
ids.setSeq(1000);
ids.next();   // "Q1000"
ids.next();   // "Q1001"

타이머 · 이벤트처럼 나중에 불리는 함수가 그때의 값을 기억하는 것도 클로저다.

function fade(node) {
  let level = 1;
  const step = () => {
    const hex = level.toString(16);
    node.style.backgroundColor = `#FFFF${hex}${hex}`;
    if (level++ < 15) setTimeout(step, 100);
  };
  setTimeout(step, 100);
}

반복문 안의 클로저

var 는 반복 전체에 변수 하나라서, 모든 핸들러가 마지막 값을 본다. let 은 반복마다 새 변수를 만든다.

for (var i = 0; i < nodes.length; i++) {
  nodes[i].onclick = () => alert(i);    // 모두 nodes.length 가 나온다
}
for (let i = 0; i < nodes.length; i++) {
  nodes[i].onclick = () => alert(i);    // 0, 1, 2 …
}
  • let 이 없던 때는 즉시 실행 함수로 반복마다 값을 붙잡았다: nodes[i].onclick = (function (i) { return () => alert(i); })(i);

즉시 실행 함수

만들자마자 부르는 함수다. 모듈이 없던 때 전역을 더럽히지 않으려고 파일 전체를 감쌌다.

(function () {
  const local = 1;   // 밖에서 보이지 않는다
  ...
})();
  • 지금은 ES 모듈 파일이 그 자체로 범위를 가진다. 모듈 에 정리했다

고차 함수

함수를 받거나 돌려주는 함수다. 조건 · 키 이름을 바꿔 끼울 수 있게 된다.

const complement = pred => (...args) => !pred(...args);    // 조건 뒤집기
const plucker = field => obj => obj?.[field];              // 속성 꺼내는 함수
const always = value => () => value;                       // 늘 같은 값

const isEven = n => n % 2 === 0;
[1, 2, 3, 4].filter(complement(isEven));    // [1, 3]
books.map(plucker("title"));                // 제목 목록

부분 적용 · 커링

인자 일부를 미리 채운 새 함수를 만든다.

const partial = (fn, ...preset) => (...later) => fn(...preset, ...later);
const add10 = partial(add, 10);
add10(5);                       // 15

const add1 = add.bind(null, 1); // bind 로도 앞쪽 인자를 채운다
add1(6);                        // 7

const curry = a => b => c => a + b + c;   // 인자를 하나씩 받는다
curry(1)(2)(3);                 // 6

메모이제이션

한 번 계산한 값을 클로저 안의 표에 넣어 두고 다시 쓴다.

function memoize(fn) {
  const cache = new Map();
  return arg => {
    if (!cache.has(arg)) cache.set(arg, fn(arg));
    return cache.get(arg);
  };
}

const fibonacci = memoize(n => n < 2 ? n : fibonacci(n - 1) + fibonacci(n - 2));
fibonacci(50);   // 12586269025 — 바로 나온다

예외

throw 는 아무 값이나 던질 수 있지만 Error 를 던진다. 호출 스택이 같이 남는다.

function add(a, b) {
  if (typeof a !== "number" || typeof b !== "number")
    throw new TypeError("add needs numbers");
  return a + b;
}

try {
  add("seven");
} catch (e) {
  console.error(`${e.name}: ${e.message}`);   // TypeError: add needs numbers
} finally {
  ...
}

throw new Error("저장 실패", { cause: err });   // 원래 오류를 붙여 다시 던진다
