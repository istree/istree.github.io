JavaScript 객체는 이름 · 값 쌍의 묶음이고, 저마다 프로토타입(다른 객체) 하나를 가리킨다. 객체에 없는 속성은 프로토타입에서 찾고, 거기도 없으면 그 프로토타입에서 찾는다(프로토타입 체인). class 문법도 이 위에서 돈다.

프로토타입 체인

const mammal = {
  says() { return this.saying ?? ""; },
  getName() { return this.name; },
};

const cat = Object.create(mammal);   // 프로토타입이 mammal 인 새 객체
cat.name = "Henrietta";
cat.saying = "meow";

cat.says();                          // "meow" — cat 에 없어서 mammal 에서 찾았다
Object.getPrototypeOf(cat) === mammal;   // true
Object.hasOwn(cat, "says");          // false — 자기 속성이 아니다
  • 속성을 읽을 때만 체인을 따라간다. 쓰면 늘 그 객체 자신에 생긴다
  • for...in 은 체인의 속성까지 돈다. 자기 속성만 돌려면 Object.keys · Object.entries 를 쓴다
  • 프로토타입이 없는 객체는 Object.create(null) 로 만든다. 키 · 값 표로만 쓸 때 toString 같은 이름과 부딪히지 않는다. 표로 쓸 때는 Map 이 더 낫다

생성자 함수

class 가 없던 때의 방식이다. new 로 부르면 새 객체의 프로토타입이 함수.prototype 이 된다.

function Person(name) {
  // this = Object.create(Person.prototype)
  this.name = name;
  // return this
}
Person.prototype.say = function () {      // 메서드는 prototype 에 하나만 둔다
  return "I am " + this.name;
};

const adam = new Person("Adam");
adam.say();                               // "I am Adam"
adam instanceof Person;                   // true
  • 생성자 안에서 this.say = function … 으로 두면 객체마다 함수가 새로 만들어진다
  • new 를 빼먹으면 this 가 전역(또는 undefined) 이 된다. 예전에는 if (!(this instanceof Person)) return new Person(name); 으로 막았다. class 는 new 없이 부르면 오류가 난다

상속은 프로토타입끼리 연결한다.

function Cat(name) {
  Person.call(this, name);                               // 부모 생성자
}
Cat.prototype = Object.create(Person.prototype);         // 체인 연결
Cat.prototype.constructor = Cat;

class

위 생성자 · 프로토타입을 한 덩어리로 쓰는 문법이다. 지금은 이것을 쓴다.

class Mammal {
  #saying;                          // 비공개 필드 — 클래스 밖에서 읽을 수 없다
  static count = 0;                 // 클래스에 붙는 값

  constructor(name, saying = "") {
    this.name = name;
    this.#saying = saying;
    Mammal.count++;
  }
  says() { return this.#saying; }
  get label() { return `${this.name} (${this.says()})`; }   // 속성처럼 읽는다
}

class Cat extends Mammal {
  constructor(name) {
    super(name, "meow");            // this 를 쓰기 전에 부른다
  }
  getName() {
    return `${this.says()} ${this.name} ${this.says()}`;
  }
  says() {
    return super.says().toUpperCase();   // 부모 메서드
  }
}

const c = new Cat("Henrietta");
c.getName();    // "MEOW Henrietta MEOW"
c.label;        // "Henrietta (MEOW)"
  • 메서드도 this 규칙은 같다. 콜백으로 넘기면 끊어진다. 함수와 클로저 의 "this" 에 정리했다
  • 핸들러로 자주 넘기는 메서드는 화살표 필드로 둔다: onClick = () => { this.… };

함수로 객체 만들기

핵심 가이드가 권하는 방식이다. new · this · prototype 없이, 함수가 객체를 만들어 돌려준다. 비공개 상태는 클로저에 둔다.

function mammal(spec) {
  const that = {};
  that.getName = () => spec.name;
  that.says = () => spec.saying ?? "";
  return that;
}

function cat(spec) {
  const that = mammal({ saying: "meow", ...spec });
  const superGetName = that.getName;          // 부모 메서드를 붙잡아 둔다
  that.getName = () => `${that.says()} ${superGetName()} ${that.says()}`;
  that.purr = n => Array(n).fill("r").join("-");
  return that;
}

const myCat = cat({ name: "Henrietta" });
myCat.getName();   // "meow Henrietta meow"
myCat.purr(3);     // "r-r-r"
  • this 를 쓰지 않으므로 메서드를 떼어 넘겨도 된다
  • 객체마다 함수를 새로 만든다. 객체가 아주 많으면 class 가 메모리를 덜 쓴다

기능 섞기

상속 대신, 있는 객체에 기능을 덧붙인다. 핵심 가이드의 eventuality 는 아무 객체에나 이벤트 기능을 붙인다.

function eventuality(that) {
  const registry = new Map();
  that.on = (type, handler) => {
    if (!registry.has(type)) registry.set(type, []);
    registry.get(type).push(handler);
    return that;
  };
  that.fire = (type, ...args) => {
    for (const handler of registry.get(type) ?? []) handler.apply(that, args);
    return that;
  };
  return that;
}

const box = eventuality({ name: "box" });
box.on("move", (x, y) => console.log(x, y)).fire("move", 10, 20);
  • 브라우저 · Node.js 에는 이미 EventTarget · EventEmitter 가 있다. 직접 만들 필요가 있을 때 이 틀을 쓴다

복사 · 비교

const shallow = { ...obj };                 // 얕은 복사 — 안쪽 객체는 같이 쓴다
const deep = structuredClone(obj);          // 깊은 복사 (함수 · DOM 은 안 된다)
Object.assign(target, a, b);                // a, b 의 속성을 target 에 덮어쓴다

const frozen = Object.freeze({ x: 1 });     // 속성을 바꿀 수 없다 (얕게)
{ a: 1 } === { a: 1 };                      // false — 객체는 같은 것인지만 비교한다

JSON

const text = JSON.stringify({ name: "a", tags: ["x"] });   // '{"name":"a","tags":["x"]}'
JSON.stringify(obj, null, 2);                              // 들여쓰기 2 칸
JSON.stringify({ when: new Date(0) });                     // 날짜는 ISO 문자열이 된다
JSON.parse(text);                                          // 객체로
JSON.parse(text, (key, value) => key === "when" ? new Date(value) : value);  // 되살리기
  • undefined · 함수 · Symbol 속성은 빠진다. Map · Set 은 {} 가 된다
  • 객체에 toJSON() 메서드가 있으면 그 결과를 문자열로 바꾼다
  • 받은 JSON 을 eval 로 읽지 않는다
