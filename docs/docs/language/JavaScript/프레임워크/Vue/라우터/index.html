Vue Router 는 주소에 맞는 컴포넌트를 <RouterView> 자리에 넣는다. 쪽을 옮겨도 페이지를 새로 읽지 않고 컴포넌트만 바꾼다. npm create vue 에서 Router 를 선택하면 설정까지 만든다.

라우트

// router/index.js
import { createRouter, createWebHistory } from "vue-router";

export default createRouter({
  history: createWebHistory(),
  routes: [
    { path: "/", component: () => import("../views/Home.vue") },          // 쓸 때 읽는다
    { path: "/users/:id", name: "user", component: () => import("../views/User.vue"),
      props: route => ({ id: Number(route.params.id) }) },                  // :id 를 숫자 props 로
    {
      path: "/settings", component: () => import("../views/Settings.vue"),
      children: [                                                           // Settings 안의 RouterView 에
        { path: "", component: () => import("../views/Profile.vue") },
        { path: "password", component: () => import("../views/Password.vue") },
      ],
    },
    { path: "/:pathMatch(.*)*", component: () => import("../views/NotFound.vue") },   // 나머지 모두
  ],
});
// main.js
import router from "./router";
createApp(App).use(router).mount("#app");
  • props: true 로 주면 :id 를 그대로 넘긴다. 주소 값은 문자열이라 id: Number 로 받으면 경고가 난다
  • () => import(…) 로 주면 그 쪽에 처음 갈 때 파일을 읽는다. 빌드에서 쪽마다 파일이 나뉜다

링크 · 옮기기

<script setup>
import { useRouter, useRoute } from "vue-router";

const router = useRouter();
const route = useRoute();                       // 지금 주소 — route.params.id · route.query.q
const go = id => router.push({ name: "user", params: { id } });
</script>

<template>
  <RouterLink to="/">처음</RouterLink>
  <RouterLink :to="{ name: 'user', params: { id: 3 } }">사용자</RouterLink>
  <RouterLink :to="{ path: '/search', query: { q: 'vue' } }">찾기</RouterLink>   <!-- /search?q=vue -->
  <RouterView />                                <!-- 주소에 맞는 컴포넌트가 들어간다 -->
</template>
  • 지금 주소의 링크에는 router-link-active 클래스가 붙는다. 메뉴를 칠할 때 쓴다
  • router.replace(…) 는 방문 기록을 남기지 않고 옮긴다. router.back() 은 뒤로 간다
  • 같은 컴포넌트에서 :id 만 바뀌면 컴포넌트를 다시 만들지 않는다. props 를 watch 로 보고 다시 받는다

가드

옮기기 전에 검사한다. 로그인이 필요한 쪽에 쓴다.

router.beforeEach(to => {
  if (to.meta.requiresAuth && !isLoggedIn()) {
    return { path: "/login", query: { redirect: to.fullPath } };   // 다른 곳으로
  }
  // 아무것도 돌려주지 않으면 그대로 간다, false 는 막는다
});

// 라우트에 표시
{ path: "/admin", component: Admin, meta: { requiresAuth: true } }
  • 예전(Vue Router 3) 에는 세 번째 인자 next 를 불러야 넘어갔다. 지금은 돌려주는 값으로 정한다

주소 방식

  • createWebHistory 는 /users/3 처럼 보통 주소를 쓴다. 서버가 모르는 주소에도 index.html 을 돌려줘야 한다
  • 정적 호스팅에서 그렇게 하지 못하면 createWebHashHistory(/#/users/3) 를 쓴다. Electron 처럼 file:// 로 여는 앱도 이것을 쓴다
