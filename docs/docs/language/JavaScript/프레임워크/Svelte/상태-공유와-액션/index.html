여러 컴포넌트가 같은 값을 쓸 때는 .svelte.js 파일에 $state 를 두고 import 한다. 컴포넌트 트리 아래로만 내려 줄 값은 context 로 넘긴다. 요소에 붙는 동작(포커스 · 바깥 클릭 · 외부 라이브러리) 은 attachment · 액션으로, 나타나고 사라지는 움직임은 transition 으로 붙인다.

.svelte.js 로 상태 공유

runes 는 이름이 .svelte.js(.svelte.ts) 인 파일에서도 쓴다. 모듈은 앱에 하나뿐이므로 거기 둔 $state 를 import 한 곳이 모두 같은 값을 본다.

// src/lib/cart.svelte.js
export const cart = $state({ items: [] });

export function addItem(name, price) {
  cart.items.push({ name, price });
}

export function total() {
  return cart.items.reduce((sum, i) => sum + i.price, 0);
}
<!-- 어느 컴포넌트에서든 -->
<script>
  import { cart, addItem, total } from "./lib/cart.svelte.js";
</script>

<button onclick={() => addItem("펜", 1500)}>담기</button>
<p>{cart.items.length} 개 · {total()} 원</p>
  • 내보낸 $state 에 새 값을 대입하면 Cannot export state from a module if it is reassigned 로 빌드가 멈춘다. export let count = $state(0) 대신 객체로 두고 속성을 바꾸거나, 값을 돌려주는 함수를 내보낸다
  • 계산한 값은 $derived 를 내보내지 않고 함수(total()) 로 내보낸다. 화면에서 부르면 그 안에서 읽은 $state 를 따라 다시 그린다

클래스의 필드에도 $state · $derived 를 쓴다. 같은 모양의 상태를 여러 개 만들 때 쓴다.

// src/lib/counter.svelte.js
export class Counter {
  count = $state(0);
  double = $derived(this.count * 2);
  inc = () => { this.count++; };        // 화살표 필드 — onclick={c.inc} 로 넘겨도 this 가 맞다
}
<script>
  import { Counter } from "./lib/counter.svelte.js";
  const c = new Counter();
</script>

<button onclick={c.inc}>{c.count} · {c.double}</button>
  • 서버 렌더링(SvelteKit) 에서는 모듈의 상태를 모든 요청이 같이 쓴다. 사용자마다 다른 값은 모듈에 두지 않고 context 로 넘긴다

context

부모가 set 한 값을 그 아래 어느 깊이의 자식이든 get 한다. props 로 여러 단계를 거쳐 내려 주지 않아도 된다.

// src/lib/theme.js
import { createContext } from "svelte";
export const [getTheme, setTheme] = createContext();
<!-- App.svelte -->
<script>
  import { setTheme } from "./lib/theme.js";
  const theme = $state({ mode: "dark" });
  setTheme(theme);                          // 반응형 객체를 넘긴다
</script>

<button onclick={() => (theme.mode = theme.mode === "dark" ? "light" : "dark")}>바꾸기</button>
<Page />
<!-- 아래 어딘가의 컴포넌트 -->
<script>
  import { getTheme } from "./lib/theme.js";
  const theme = getTheme();
</script>

<p>{theme.mode}</p>                         <!-- 부모에서 바꾸면 따라 바뀐다 -->
  • set · get 은 컴포넌트를 만드는 중(<script> 맨 위) 에만 부른다. 이벤트 처리 함수 안에서 부르면 lifecycle_outside_component 오류가 난다
  • createContext 는 Svelte 5.40 부터 있다. 그 전에는 setContext("theme", theme) · getContext("theme") 처럼 키로 했다

attachment · 액션

요소가 화면에 붙을 때 그 요소로 할 일을 함수 하나로 묶는다. 돌려준 함수는 요소가 떨어질 때 불린다.

<script>
  function clickOutside(node) {
    const handler = e => {
      if (!node.contains(e.target)) node.dispatchEvent(new CustomEvent("outside"));
    };
    document.addEventListener("click", handler, true);
    return () => document.removeEventListener("click", handler, true);
  }
  let open = $state(true);
</script>

{#if open}
  <div class="menu" {@attach clickOutside} onoutside={() => (open = false)}>메뉴</div>
{/if}
  • {@attach fn} 은 Svelte 5.29 부터 있다. 안에서 읽은 $state 가 바뀌면 정리하고 다시 돈다
  • 차트 · 지도 같은 외부 라이브러리를 요소에 붙일 때도 이 틀을 쓴다: 붙을 때 만들고, 돌려준 함수에서 destroy() 한다

예전 방식인 액션(use:) 도 그대로 쓴다. 인자를 받고, 인자가 바뀌면 update 가 불린다.

<script>
  function autofocus(node) {
    node.focus();
  }
  function tooltip(node, text) {
    node.title = text;
    return {
      update(next) { node.title = next; },
      destroy() {},
    };
  }
</script>

<input use:autofocus />
<span use:tooltip={"도움말"}>?</span>

transition

{#if} · {#each} 로 나타나고 사라질 때 움직임을 준다.

<script>
  import { fade, fly, slide } from "svelte/transition";
  import { flip } from "svelte/animate";
  let visible = $state(true);
</script>

{#if visible}
  <p transition:fade={{ duration: 200 }}>들어오고 나간다</p>
  <p in:fly={{ y: 20 }} out:fade>들어올 때 · 나갈 때 다르게</p>
{/if}

{#each items as item (item.id)}
  <li animate:flip={{ duration: 300 }} transition:slide>{item.name}</li>   <!-- 자리가 바뀌면 미끄러진다 -->
{/each}
  • animate: 는 key 가 있는 {#each} 바로 안의 요소에만 쓴다
  • 사용자가 움직임 줄이기를 켰으면(prefers-reduced-motion) duration: 0 으로 준다

예전에는

  • 공유 상태는 store(svelte/store 의 writable · derived) 로 했다. 컴포넌트에서 $cart 처럼 $ 를 붙이면 구독 · 해지를 알아서 했다. Svelte 5 에서도 쓰지만, 새 코드는 .svelte.js 의 $state 로 한다
// stores.js — Svelte 3 · 4
import { writable, derived } from "svelte/store";
export const cart = writable([]);
export const total = derived(cart, $c => $c.reduce((s, i) => s + i.price, 0));
// 컴포넌트에서: {$total} · cart.update(c => [...c, item])
