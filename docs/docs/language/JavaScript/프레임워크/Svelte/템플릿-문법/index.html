Svelte 의 마크업은 HTML 에 {} 식과 블록({#if} · {#each} · {#await}) 을 더한 것이다. 이벤트는 onclick 같은 속성으로, 양방향 연결은 bind: 로 쓴다. 아래는 할 일 목록 하나로 문법을 모두 쓴다.

<script>
  let todos = $state([
    { id: 1, text: "장보기", done: false },
    { id: 2, text: "운동", done: true },
  ]);
  let draft = $state("");
  let filter = $state("all");

  const shown = $derived(
    filter === "all" ? todos : todos.filter(t => t.done === (filter === "done")),
  );

  function add(e) {
    e.preventDefault();
    if (!draft.trim()) return;
    todos.push({ id: Date.now(), text: draft, done: false });
    draft = "";
  }
</script>

<form onsubmit={add}>
  <input bind:value={draft} placeholder="할 일" />
  <button>추가</button>
</form>

<select bind:value={filter}>
  <option value="all">모두</option>
  <option value="done">한 것</option>
  <option value="todo">할 것</option>
</select>

{#if shown.length === 0}
  <p>없다</p>
{:else}
  <ul>
    {#each shown as todo, i (todo.id)}
      <li class={{ done: todo.done }}>
        <input type="checkbox" bind:checked={todo.done} />
        {i + 1}. {todo.text}
      </li>
    {/each}
  </ul>
{/if}

<p>남은 일 {todos.filter(t => !t.done).length} 개</p>

식 · 속성

<p>{user.name.toUpperCase()}</p>            <!-- 식 하나 -->
<img src={url} alt="" />                    <!-- 속성 값 -->
<img {src} alt="" />                        <!-- src={src} 를 줄인 것 -->
<a href="/users/{id}">보기</a>              <!-- 문자열 안에 끼우기 -->
<button disabled={!agree}>다음</button>     <!-- false · null · undefined 면 속성이 빠진다 -->
<Card {...props} />                         <!-- 객체를 속성으로 펼치기 -->
<p>{@html html}</p>                         <!-- 문자열을 HTML 로. 남이 넣은 값은 넣지 않는다 -->
  • {} 안의 값은 글자로 들어간다. <b> 를 넣어도 태그가 되지 않는다

class · style

<li class={{ done: todo.done, urgent }}>       <!-- 참인 키만 붙는다 -->
<li class={["item", todo.done && "done"]}>     <!-- 배열 — 거짓 값은 빠진다 -->
<p style:color={color}>색</p>                  <!-- style="color: tomato;" -->
<p style:color>색</p>                          <!-- 변수 이름이 같으면 줄인다 -->
  • 예전(Svelte 5.16 아래) 에는 class:done={todo.done} 으로 하나씩 붙였다. 지금도 된다

조건 · 반복

{#if user}
  <p>{user.name}</p>
{:else if loading}
  <p>불러오는 중</p>
{:else}
  <p>없다</p>
{/if}

{#each items as item, i (item.id)}     <!-- (item.id) 는 key — 자리가 바뀌어도 요소를 다시 만들지 않는다 -->
  <li>{i}: {item.name}</li>
{:else}
  <li>비었다</li>                      <!-- 배열이 비었을 때 -->
{/each}

{#each { length: 3 }, i}<span>{i}</span>{/each}   <!-- 배열 없이 세 번 -->
  • 순서가 바뀌거나 중간에 끼우고 빼는 목록에는 key 를 준다. key 가 없으면 자리마다 내용만 바꾼다

비동기 값

Promise 를 바로 기다린다.

<script>
  const promise = fetch("/api/items").then(r => r.json());
</script>

{#await promise}
  <p>불러오는 중</p>
{:then items}
  <p>{items.length} 개</p>
{:catch err}
  <p>{err.message}</p>
{/await}
  • promise 를 새 Promise 로 바꾸면 처음부터 다시 기다린다. 버튼으로 다시 불러올 때는 promise = load() 로 바꾼다($state 로 둔다)

이벤트

on 뒤에 이벤트 이름을 붙인 속성에 함수를 준다.

<button onclick={() => count++}>+</button>
<button onclick={save}>저장</button>                 <!-- 함수 이름만 -->
<form onsubmit={e => { e.preventDefault(); add(); }}>
<input onkeydown={e => e.key === "Enter" && add()} />
<svelte:window onkeydown={handleKey} />                <!-- window 이벤트 -->
  • 예전에는 on:click={…} 과 수식어(on:submit|preventDefault) 를 썼다. Svelte 5 에서는 onclick 으로 쓰고, 수식어 대신 함수 안에서 e.preventDefault() 를 부른다

bind

입력 요소의 값과 변수를 양쪽으로 연결한다.

<input bind:value={draft} />                           <!-- 문자열 -->
<input type="number" bind:value={qty} />               <!-- 숫자로 들어온다 -->
<input type="checkbox" bind:checked={agree} />
<input type="radio" bind:group={size} value="M" />     <!-- 같은 group 끼리 하나 -->
<select bind:value={filter}>…</select>
<textarea bind:value={memo}></textarea>
<div bind:clientWidth={w}>…</div>                      <!-- 크기 읽기 (읽기만) -->
<canvas bind:this={canvas}></canvas>                   <!-- 요소 자체 -->
  • 연결할 변수는 $state 로 둔다
  • bind:this 로 받은 요소는 화면에 붙은 뒤(onMount · $effect 안) 에 쓴다
