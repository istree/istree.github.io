Node 에는 테스트 실행기(node:test) 와 단언(node:assert) 이 들어 있다. node --test 가 테스트 파일을 찾아 돌리고, 가짜 함수 · 가짜 타이머 · 커버리지도 된다. 작은 프로젝트는 Jest · Vitest 를 설치하지 않고 이것으로 한다.

테스트 쓰기

// src/math.js
export function add(a, b) { return a + b; }
export function div(a, b) {
  if (b === 0) throw new RangeError("0 으로 나눌 수 없다");
  return a / b;
}
// test/math.test.js
import { test, describe, it } from "node:test";
import assert from "node:assert/strict";
import { add, div } from "../src/math.js";

test("add", () => {
  assert.equal(add(1, 2), 3);
});

describe("div", () => {                             // 묶음
  it("나눈다", () => assert.equal(div(6, 3), 2));
  it("0 이면 던진다", () => {
    assert.throws(() => div(1, 0), RangeError);
    assert.throws(() => div(1, 0), { message: /0 으로/ });
  });
});

test.skip("나중에", () => {});
test.todo("할 일");
{
  "type": "module",
  "scripts": { "test": "node --test" }
}
$ npm test

  ✔ 0 이면 던진다 (0.213166ms)

  • test 와 it, describe 와 suite 는 같다. Jest 처럼 쓰려면 describe · it 을 쓴다
  • 함수가 예외를 던지거나 돌려준 Promise 가 거부되면 실패한다
  • 실패가 하나라도 있으면 종료 코드가 1 이다. CI 가 이것으로 멈춘다

단언

node:assert/strict 를 쓴다. === 와 같은 엄격한 비교를 한다.

assert.equal(actual, expected);                     // ===
assert.notEqual(a, b);
assert.deepEqual({ a: [1, 2] }, { a: [1, 2] });     // 내용이 같다
assert.ok(value);                                   // 참 같은 값
assert.match("hello", /ell/);
assert.throws(() => fn(), TypeError);               // 던진다 — 클래스 · 정규식 · { message } 로 확인
await assert.rejects(promise, /HTTP 404/);          // 거부된다 — await 를 꼭 붙인다

실패하면 다른 곳을 보여 준다.

✖ 틀린 것 (1.122958ms)
  AssertionError [ERR_ASSERTION]: Expected values to be strictly deep-equal:
  + actual - expected

    {
      a: 1,
      b: [
        1,
  +     2
  -     3
      ]
    }
  • node:assert (strict 가 아닌 것) 의 equal 은 == 로 비교해 1 과 "1" 을 같다고 본다. 쓰지 않는다

비동기

async 함수로 쓰면 끝날 때까지 기다린다.

// src/math.js — get 을 바꿔 끼울 수 있게 인자로 받는다
export async function fetchUser(id, get = fetch) {
  const res = await get(`https://api.example.com/users/${id}`);
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return res.json();
}
import { mock } from "node:test";

describe("fetchUser", () => {
  it("JSON 을 돌려준다", async () => {
    const get = mock.fn(async () => ({ ok: true, json: async () => ({ id: 1, name: "geon" }) }));
    assert.deepEqual(await fetchUser(1, get), { id: 1, name: "geon" });
    assert.equal(get.mock.callCount(), 1);
    assert.equal(get.mock.calls[0].arguments[0], "https://api.example.com/users/1");
  });

  it("404 면 거부한다", async () => {
    const get = async () => ({ ok: false, status: 404 });
    await assert.rejects(fetchUser(1, get), /HTTP 404/);
  });
});
  • mock.fn(구현) 은 부른 횟수 · 인자를 .mock.calls 에 남기는 가짜 함수다
  • 객체의 메서드를 바꿔 끼울 때는 mock.method(obj, "name", 구현) 을 쓴다. 테스트 안에서 t.mock 으로 만들면 테스트가 끝날 때 원래대로 돌아온다

가짜 타이머

setTimeout 을 실제로 기다리지 않고 시간을 앞으로 돌린다.

// src/math.js
export function debounce(fn, ms) {
  let t;
  return (...args) => { clearTimeout(t); t = setTimeout(() => fn(...args), ms); };
}
test("debounce", (t) => {
  t.mock.timers.enable({ apis: ["setTimeout"] });
  const fn = t.mock.fn();
  const d = debounce(fn, 100);

  d(1); d(2); d(3);
  t.mock.timers.tick(99);
  assert.equal(fn.mock.callCount(), 0);
  t.mock.timers.tick(1);
  assert.equal(fn.mock.callCount(), 1);
  assert.deepEqual(fn.mock.calls[0].arguments, [3]);   // 마지막 것만
});

서버 테스트

node:http 로 만든 서버(src/server.js) 를 테스트에서 띄우려면, 서버를 내보내고 직접 실행할 때만 listen 하게 바꾼다.

// src/server.js 끝
export default server;
if (import.meta.main) server.listen(3000);            // node src/server.js 로 실행할 때만
// test/server.test.js
import { describe, it, before, after } from "node:test";
import assert from "node:assert/strict";
import { once } from "node:events";
import server from "../src/server.js";

describe("API", () => {
  let base;
  before(async () => {
    server.listen(0);                                 // 비어 있는 포트 아무거나
    await once(server, "listening");
    base = `http://localhost:${server.address().port}`;
  });
  after(() => server.close());                        // 닫지 않으면 테스트가 끝나지 않는다

  it("목록", async () => {
    const res = await fetch(`${base}/api/notes`);
    assert.equal(res.status, 200);
    assert.deepEqual(await res.json(), [{ id: 1, text: "hello" }]);
  });

  it("잘못된 JSON 은 400", async () => {
    const res = await fetch(`${base}/api/notes`, { method: "POST", body: "nope" });
    assert.equal(res.status, 400);
  });
});
  • before · after 는 묶음에서 한 번, beforeEach · afterEach 는 테스트마다 돈다
  • 포트 0 을 주면 OS 가 빈 포트를 준다. 테스트 파일이 함께 돌아도 부딪치지 않는다

실행 옵션

node --test                                    # 테스트 파일을 모두 찾아 실행
node --test "test/**/*.test.js"                # 이 패턴만
node --test --test-name-pattern="div"          # 이름이 맞는 테스트만
node --test --watch                            # 고칠 때마다 다시
node --test --test-reporter=dot                # 짧게 — spec(기본) · tap · junit · lcov
node --test --experimental-test-coverage       # 커버리지 표
node --test --experimental-test-coverage --test-coverage-lines=90   # 90% 밑이면 실패
ℹ file      | line % | branch % | funcs % | uncovered lines
ℹ  math.js  |  84.21 |   100.00 |   85.71 | 17-19

  • 패턴을 주지 않으면 *.test.* · *-test.* · *_test.* · test-*.* 꼴의 파일과 test/ 폴더 안의 파일을 찾는다. *.spec.js 는 찾지 않으므로 패턴을 준다
  • 파일은 따로 프로세스에서 함께 돈다. 한 파일 안의 테스트는 차례로 돈다
  • .ts 테스트도 그대로 돈다. 타입은 지우기만 한다
  • it.only 로 표시한 것만 돌리려면 --test-only 를 준다
