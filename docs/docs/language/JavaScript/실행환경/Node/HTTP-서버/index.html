Node 는 node:http 로 웹 서버를 띄운다. 요청마다 함수 하나가 불리고, 그 안에서 주소 · 메서드를 보고 응답을 쓴다. 작은 API · 정적 파일 서버는 패키지 없이 만들고, 경로가 많아지면 Express 를 쓴다. 아래에서는 같은 메모 API(목록 · 검색 · 추가) 와 정적 파일 서비스를 두 가지로 만든다.

node:http

// server.js — node --watch server.js
import http from "node:http";
import { readFile } from "node:fs/promises";
import { json } from "node:stream/consumers";
import path from "node:path";

const notes = [{ id: 1, text: "hello" }];
const root = path.join(import.meta.dirname, "public");
const types = { ".html": "text/html; charset=utf-8", ".css": "text/css", ".js": "text/javascript", ".svg": "image/svg+xml", ".png": "image/png" };

function send(res, status, body) {
  res.writeHead(status, { "Content-Type": "application/json; charset=utf-8" });
  res.end(JSON.stringify(body));
}

const server = http.createServer(async (req, res) => {
  const url = new URL(req.url, `http://${req.headers.host}`);   // req.url 은 "/api/notes?q=a" 꼴
  try {
    if (url.pathname === "/api/notes" && req.method === "GET") {
      const q = url.searchParams.get("q");
      return send(res, 200, q ? notes.filter(n => n.text.includes(q)) : notes);
    }
    if (url.pathname === "/api/notes" && req.method === "POST") {
      const body = await json(req);                               // 본문을 다 읽어 JSON.parse
      if (typeof body.text !== "string") return send(res, 400, { error: "text 가 필요하다" });
      const note = { id: notes.length + 1, text: body.text };
      notes.push(note);
      return send(res, 201, note);
    }
    if (url.pathname.startsWith("/api/")) return send(res, 404, { error: "not found" });

    // 정적 파일 — public/ 밖은 내주지 않는다
    const file = path.join(root, url.pathname === "/" ? "index.html" : decodeURIComponent(url.pathname));
    if (!file.startsWith(root + path.sep)) return send(res, 403, { error: "forbidden" });
    const data = await readFile(file);
    res.writeHead(200, { "Content-Type": types[path.extname(file)] ?? "application/octet-stream" });
    res.end(data);
  } catch (e) {
    if (e.code === "ENOENT") return send(res, 404, { error: "not found" });
    if (e instanceof SyntaxError) return send(res, 400, { error: "JSON 이 아니다" });
    console.error(e);
    send(res, 500, { error: "server error" });
  }
});

server.listen(3000, () => console.log("http://localhost:3000"));
$ curl localhost:3000/api/notes
[{"id":1,"text":"hello"}]
$ curl -X POST localhost:3000/api/notes -H 'Content-Type: application/json' -d '{"text":"world"}'
{"id":2,"text":"world"}                       # 201
$ curl -X POST localhost:3000/api/notes -d 'nope'
{"error":"JSON 이 아니다"}                    # 400
$ curl 'localhost:3000/api/notes?q=wor'
[{"id":2,"text":"world"}]
$ curl --path-as-is 'localhost:3000/..%2fpackage.json'
{"error":"forbidden"}                         # 403
  • res.end() 를 부르지 않으면 응답이 끝나지 않아 브라우저가 계속 기다린다. 모든 갈래에서 부른다
  • 요청 본문은 스트림으로 온다. node:stream/consumers 의 json(req) · text(req) 로 한 번에 읽는다
  • 정적 파일 경로는 decodeURIComponent 로 %20 을 풀고, 합친 경로가 public/ 안인지 확인한다. 빼면 ..%2f 로 서버의 아무 파일이나 읽힌다
  • listen(3000) 은 모든 네트워크에서 받는다. 내 컴퓨터에서만 받으려면 listen(3000, "127.0.0.1") 로 한다
  • 브라우저에서 다른 주소(localhost:5173 같은 개발 서버) 로 부르면 CORS 에 막힌다. 응답에 Access-Control-Allow-Origin 헤더를 넣거나, 개발 서버의 프록시(Vite 의 server.proxy) 를 쓴다

Express

같은 서버를 Express 5 로 만든다. 경로마다 함수를 달고, 본문 읽기 · 정적 파일 · 오류 처리를 미들웨어로 넣는다.

npm install express
// app.js
import express from "express";

const app = express();
const notes = [{ id: 1, text: "hello" }];

app.use(express.json());                     // Content-Type 이 JSON 인 본문 → req.body
app.use(express.static("public"));           // public/ 의 파일 — ../ 는 막는다

app.get("/api/notes", (req, res) => {
  const { q } = req.query;
  res.json(q ? notes.filter(n => n.text.includes(q)) : notes);
});

app.get("/api/notes/:id", (req, res) => {    // /api/notes/1 → req.params.id 는 "1" (문자열)
  const note = notes.find(n => n.id === Number(req.params.id));
  if (!note) return res.status(404).json({ error: "not found" });
  res.json(note);
});

app.post("/api/notes", (req, res) => {
  if (typeof req.body?.text !== "string") return res.status(400).json({ error: "text 가 필요하다" });
  const note = { id: notes.length + 1, text: req.body.text };
  notes.push(note);
  res.status(201).json(note);
});

app.get("/api/fail", async () => {
  throw new Error("boom");                   // async 함수의 오류도 아래로 간다
});

app.use((err, req, res, next) => {           // 오류 처리 — 인자가 넷이어야 한다
  console.error(err.message);
  res.status(err.status ?? 500).json({ error: err.message });
});

app.listen(3000, () => console.log("http://localhost:3000"));
  • Express 5 는 async 함수가 던진 오류 · 거부를 오류 처리 미들웨어로 넘긴다. Express 4 는 try 로 잡아 next(e) 를 불러야 했다
  • 본문이 JSON 이 아니면 express.json() 이 400 과 Unexpected token 'n', "nope" is not valid JSON 을 오류 처리로 넘긴다. Content-Type 이 JSON 이 아니면 req.body 가 undefined 다
  • 미들웨어는 적은 순서대로 돈다. 오류 처리는 맨 뒤에 둔다
  • 맞는 경로가 없으면 Express 가 Cannot GET /zzz 를 404 로 낸다

fetch 로 부르기

다른 스크립트 · 테스트에서 서버를 부를 때는 전역 fetch 를 쓴다.

const res = await fetch("http://localhost:3000/api/notes", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ text: "from fetch" }),
});
console.log(res.status, await res.json());   // 201 { id: 2, text: 'from fetch' }
